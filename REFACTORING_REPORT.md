### 1. Автоматическое определение платформы
```python
def auto_detect_platform(message: types.Message) -> str:
    """Автоматически определяет платформу пользователя"""
    user = message.from_user
    
    # Анализ языка интерфейса
    if user.language_code in ['zh', 'zh-cn', 'zh-tw', 'ja', 'ko']:
        return "ios"
    
    # Анализ имени пользователя
    if user.username and any(char in user.username for char in ['📱', '🍎', '💻']):
        return "ios"
    
    return "desktop"  # по умолчанию
```
